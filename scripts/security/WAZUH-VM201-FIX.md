```bash
#!/bin/bash
echo "🚀 CORTEX FIX: Wazuh Dashboard v4.11.2 → v4.14.1"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

# Show current state
echo -e "\n📊 BEFORE:"
echo "API Version:"
/var/ossec/bin/wazuh-control info 2>/dev/null | grep VERSION || echo "  (checking...)"
echo "Dashboard Version:"
dpkg -l 2>/dev/null | grep wazuh-dashboard | awk '{print "  "$2" "$3}' || rpm -qa 2>/dev/null | grep wazuh-dashboard

# Detect OS and upgrade
echo -e "\n🔧 UPGRADING..."
if [ -f /etc/debian_version ]; then
    echo "  OS: Debian/Ubuntu detected"
    apt-get update -qq >/dev/null 2>&1
    apt-get install -y wazuh-dashboard=4.14.1-1
elif [ -f /etc/redhat-release ]; then
    echo "  OS: RHEL/CentOS detected"
    yum clean all -q >/dev/null 2>&1
    yum install -y wazuh-dashboard-4.14.1-1
else
    echo "  ❌ Unknown OS - cannot proceed"
    exit 1
fi

# Restart service
echo -e "\n🔄 RESTARTING SERVICE..."
systemctl restart wazuh-dashboard
sleep 5

# Verify
echo -e "\n✅ AFTER:"
echo "Dashboard Version:"
dpkg -l 2>/dev/null | grep wazuh-dashboard | awk '{print "  "$2" "$3}' || rpm -qa 2>/dev/null | grep wazuh-dashboard
echo "Service Status:"
systemctl is-active wazuh-dashboard && echo "  ✓ Running" || echo "  ⚠ Not running"

# Final check
echo -e "\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
if systemctl is-active wazuh-dashboard >/dev/null 2>&1; then
    echo "🎉 CORTEX FIX COMPLETE!"
    echo ""
    echo "Next steps:"
    echo "  1. Clear browser cache (Ctrl+Shift+Delete)"
    echo "  2. Refresh Wazuh dashboard"
    echo "  3. Version mismatch error should be GONE"
else
    echo "⚠️  Dashboard service needs attention"
    echo "Run: systemctl status wazuh-dashboard"
fi
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
```
